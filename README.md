# Quick Usage

```jsx
import { Baby } from "daneshicons";

export default function App() {
  return (
    <div className="flex items-center gap-3">
      <Baby
        size={38}
        color="#6366f1"
        strokeWidth={2}
      />

      <h1>Daneshicons Working 🚀</h1>
    </div>
  );
}
```

# ERP Dashboard Example

```jsx
import {
  Baby
} from "daneshicons";

export default function Dashboard() {
  return (
    <div
      style={{
        background: "#ffffff",
        borderRadius: "24px",
        padding: "24px",
        display: "flex",
        alignItems: "center",
        gap: "18px",
        boxShadow: "0 8px 24px rgba(0,0,0,0.08)"
      }}
    >
      <Baby size={36} color="#111827" />

      <div>
        <h2>Employee Management</h2>

        <p>
          Modern ERP dashboard powered by Daneshicons.
        </p>
      </div>
    </div>
  );
}
```
